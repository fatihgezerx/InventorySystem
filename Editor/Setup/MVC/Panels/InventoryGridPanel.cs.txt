#if HAS_INVENTORY_SYSTEM && HAS_UNIMVC && HAS_EVENT_SYSTEM && HAS_INPUT_SYSTEM
using System.Collections.Generic;
#if HAS_DOTWEEN
using DG.Tweening;
#endif
using UniMVC;
using UnityEngine;
using UnityEngine.EventSystems;
using UnityEngine.InputSystem;
using UnityEngine.UI;

namespace InventorySystem.UI
{
    /// <summary>
    /// Shows a <see cref="GridInventory"/> like a Resident Evil 4 attaché case: a background cell per grid
    /// cell, and one <see cref="InventorySlotButton"/> per placed item, sized to its rectangle with its icon
    /// turned by its rotation. A dragged item is the slot itself, held freely by the pointer over everything else;
    /// the rectangle it would land on is tinted green (fits) or red (doesn't), and the rotate input (R by default)
    /// turns it. Drop it on free cells to move it, or on a stack of the same item to merge: it then glides into its
    /// cell - or back to where it was - with DOTween (at once without it). <see cref="Organize"/> packs everything
    /// tightly.
    /// </summary>
    /// <remarks>
    /// The background cells can be placed under Cell Layer in the editor, so the grid can be seen and styled
    /// before play: every Image under Cell Layer is used as a cell, and only missing ones are made from Cell
    /// Template - e.g. after <see cref="GridInventory.Resize"/>.
    /// </remarks>
    public sealed class InventoryGridPanel : InventoryContentPanel
    {
        [Tooltip("Holds the background cells. Anchored and sized by the panel.")]
        [HideInInspector] [SerializeField] private RectTransform cellLayer;

        [Tooltip("Holds the item buttons. Anchored and sized by the panel.")]
        [HideInInspector] [SerializeField] private RectTransform itemLayer;

        [Tooltip("Holds the drop preview: one tinted rectangle. Anchored and sized by the panel.")]
        [HideInInspector] [SerializeField] private RectTransform previewLayer;

        [Tooltip("Copied for grid cells that aren't under Cell Layer yet, and for the drop preview.")]
        [HideInInspector] [SerializeField] private Image cellTemplate;
        [HideInInspector] [SerializeField] private InventorySlotButton itemTemplate;

        [Tooltip("Sized to the whole grid, so the window around it fits.")]
        [HideInInspector] [SerializeField] private LayoutElement layoutElement;

        [Min(8f)] [SerializeField] private float cellSize = 64f;
        [Min(0f)] [SerializeField] private float cellSpacing = 2f;

        [Tooltip("Gap between an item's icon and the edge of its footprint.")]
        [Min(0f)] [SerializeField] private float iconPadding = 6f;

        [SerializeField] private Color validColor = new(0.3f, 0.9f, 0.4f, 0.35f);
        [SerializeField] private Color invalidColor = new(0.95f, 0.3f, 0.3f, 0.35f);

        [Tooltip("Seconds a released item takes to glide into its cell, or back. 0: it jumps. Needs DOTween.")]
        [Min(0f)] [SerializeField] private float settleDuration = 0.12f;

        [Tooltip("The easing curve of that glide.")]
        [SerializeField] private AnimationEase settleEase = AnimationEase.OutBack;

        [Tooltip("Turns the dragged item. Empty: the project-wide action \"RotateItem\" if there is one, otherwise R.")]
        [SerializeField] private InputActionReference rotateAction;

        private readonly List<Image> _cells = new();
        private readonly List<InventorySlotButton> _items = new();
        private bool _cellsCollected;
        private Vector2Int _laidOut = new(-1, -1);
        private Image _preview;
        private InputAction _rotate;
        private bool _ownsRotate;
        private Vector2 _lastPointer;

        // The item slot the pointer holds, free over the grid; null when nothing is dragged.
        private InventorySlotButton _lifted;

        private GridInventory Grid => Inventory as GridInventory;

        private float Pitch => cellSize + cellSpacing;

        public override bool Supports(InventoryType type) => type == InventoryType.Grid;

        protected override void OnEnable()
        {
            ResolveRotateAction();
            _rotate.performed += OnRotatePerformed;
            _rotate.Enable();
            base.OnEnable();
        }

        protected override void OnDisable()
        {
            _rotate.performed -= OnRotatePerformed;
            if (_ownsRotate)
            {
                _rotate.Disable();
            }

            base.OnDisable();
        }

        private void OnDestroy()
        {
            if (_ownsRotate)
            {
                _rotate?.Dispose();
            }
        }

        #region Drawing

        protected override void Rebuild()
        {
            var grid = Grid;
            LayoutGrid(grid?.Columns ?? 0, grid?.Rows ?? 0);

            var slotCount = grid?.SlotCount ?? 0;
            for (var i = 0; i < slotCount; i++)
            {
                RefreshSlot(i);
            }

            for (var i = slotCount; i < _items.Count; i++)
            {
                _items[i].gameObject.SetActive(false);
            }

            HidePreview();
        }

        /// <summary>
        /// Sizes the panel to a <paramref name="columns"/> x <paramref name="rows"/> grid and lays out one
        /// background cell per grid cell: the cells already under Cell Layer first, then new ones from Cell
        /// Template. Runs on every redraw. Cells beyond the grid are hidden.
        /// </summary>
        public void LayoutGrid(int columns, int rows)
        {
            columns = Mathf.Max(0, columns);
            rows = Mathf.Max(0, rows);

            // Redrawn on every open: nothing needs touching while the grid keeps its size.
            if (_laidOut == new Vector2Int(columns, rows))
            {
                return;
            }

            _laidOut = new Vector2Int(columns, rows);
            var size = columns > 0 && rows > 0 ? Footprint(new Vector2Int(columns, rows)) : Vector2.zero;

            PrepareLayer(cellLayer, size);
            PrepareLayer(itemLayer, size);
            PrepareLayer(previewLayer, size);

            if (layoutElement != null)
            {
                layoutElement.preferredWidth = size.x;
                layoutElement.preferredHeight = size.y;
            }

            CollectCells();

            var cellCount = columns * rows;
            while (_cells.Count < cellCount)
            {
                var cell = Instantiate(cellTemplate, cellLayer);
                cell.raycastTarget = false;
                PrepareCellRect(cell.rectTransform);
                _cells.Add(cell);
            }

            for (var i = 0; i < _cells.Count; i++)
            {
                var show = i < cellCount;
                if (_cells[i].gameObject.activeSelf != show)
                {
                    _cells[i].gameObject.SetActive(show);
                }

                if (!show)
                {
                    continue;
                }

                var cell = new Vector2Int(i % columns, i / columns);
                _cells[i].rectTransform.anchoredPosition = CellPosition(cell);
            }
        }

        /// <summary>
        /// Packs the shown grid's items tightly (<see cref="GridInventory.Organize"/>) - e.g. for an Organize
        /// button. Returns false if there is no grid, or the packing couldn't fit every item (nothing moves then).
        /// </summary>
        public bool Organize()
        {
            CancelDrag();
            return Grid != null && Grid.Organize();
        }

        protected override void RefreshSlot(int index)
        {
            var grid = Grid;
            if (grid == null)
            {
                return;
            }

            while (_items.Count <= index)
            {
                var created = CreateSlot(itemTemplate, itemLayer, _items.Count);
                var rect = created.RectTransform;
                rect.anchorMin = rect.anchorMax = rect.pivot = new Vector2(0f, 1f);
                _items.Add(created);
            }

            var item = _items[index];
            var slot = grid.Slots[index];

            // The item held by the pointer comes down when its drop was handled, or its stack is gone; a change
            // of its amount while it is held only redraws the amount.
            var landing = _lifted == item && (slot.IsEmpty || InventoryDrag.Handled || InventoryDrag.SourcePanel != this);
            if (_lifted == item && !landing)
            {
                item.Show(slot);
                return;
            }

            if (landing)
            {
                _lifted = null;
            }

            if (slot.IsEmpty)
            {
                EndFlight(item);
                item.gameObject.SetActive(false);
                return;
            }

            var position = CellPosition(slot.Position);
            var size = Footprint(slot.Size);
            item.SetIconRotation(slot.Rotation, IconSize(slot.Item));
            item.Show(slot);
            item.gameObject.SetActive(true);

            if (landing)
            {
                Settle(item, position, size);
                return;
            }

            EndFlight(item);
            item.RectTransform.anchoredPosition = position;
            item.RectTransform.sizeDelta = size;
        }

        protected override void OnSlotCountChanged(int count)
        {
            // Trimmed slots were emptied (and hidden) by SlotChanged already; new ones arrive through it next.
            for (var i = count; i < _items.Count; i++)
            {
                _items[i].gameObject.SetActive(false);
            }
        }

        #endregion

        #region Drag and drop

        protected override void OnDragStarted(InventorySlotButton slot, InventorySlot inventorySlot, PointerEventData eventData)
        {
            _lastPointer = eventData.position;
            var offset = PointerCell(eventData.position) - inventorySlot.Position;
            var size = inventorySlot.Size;
            InventoryDrag.GrabOffset = new Vector2Int(Mathf.Clamp(offset.x, 0, size.x - 1), Mathf.Clamp(offset.y, 0, size.y - 1));

            // The slot itself follows the pointer, freely.
            EndFlight(slot);
            _lifted = slot;
            slot.SetLifted(true);
            ShapeLifted(inventorySlot);
            MoveLifted();
            UpdatePreview();
        }

        protected override void OnDragMoved(PointerEventData eventData)
        {
            _lastPointer = eventData.position;
            MoveLifted();
            UpdatePreview();
        }

        // The drag is over: an item still held wasn't taken anywhere (or the drop changed nothing), so it glides back.
        protected override void OnDragEnded()
        {
            HidePreview();
            if (_lifted == null)
            {
                return;
            }

            var item = _lifted;
            _lifted = null;

            var grid = Grid;
            if (grid == null || (uint)item.SlotIndex >= (uint)grid.SlotCount || grid.Slots[item.SlotIndex].IsEmpty)
            {
                EndFlight(item);
                return;
            }

            var slot = grid.Slots[item.SlotIndex];
            item.SetIconRotation(slot.Rotation, IconSize(slot.Item));
            item.Show(slot);
            Settle(item, CellPosition(slot.Position), Footprint(slot.Size));
        }

        protected override void DropFromSameInventory(InventorySlotButton target, PointerEventData eventData)
        {
            var grid = Grid;
            var index = InventoryDrag.SlotIndex;
            var amount = InventoryDrag.Amount;
            var rotation = InventoryDrag.Rotation;
            if (grid == null || !TryGetSlot(index, out var slot))
            {
                return;
            }

            var cell = PointerCell(eventData.position);
            var under = grid.GetSlotIndexAt(cell);
            if (under >= 0 && under != index && grid.Slots[under].Item == slot.Item)
            {
                grid.Move(index, under, amount);
                return;
            }

            var position = cell - InventoryDrag.GrabOffset;
            if (amount >= slot.Amount)
            {
                grid.MoveTo(index, position, rotation);
                return;
            }

            // Part of a stack dropped on free cells: split it off, then move the new stack there.
            var split = grid.Split(index, amount);
            if (split >= 0)
            {
                grid.MoveTo(split, position, rotation);
            }
        }

        private void OnRotatePerformed(InputAction.CallbackContext context)
        {
            if (InventoryDrag.SourcePanel != this || !TryGetSlot(InventoryDrag.SlotIndex, out var slot) || !slot.Item.IsRotatable)
            {
                return;
            }

            // The grabbed cell turns with the item, so it stays under the pointer: in a w x h box, a quarter
            // turn clockwise takes (x, y) to (h - 1 - y, x), and turning back takes it to (y, w - 1 - x).
            var item = slot.Item;
            var oldSize = item.GetSize(InventoryDrag.Rotation);
            var grab = InventoryDrag.GrabOffset;
            InventoryDrag.Rotation = item.NormalizeRotation(InventoryDrag.Rotation + 1);
            var newSize = item.GetSize(InventoryDrag.Rotation);
            var turned = InventoryDrag.Rotation == 1
                ? new Vector2Int(oldSize.y - 1 - grab.y, grab.x)
                : new Vector2Int(grab.y, oldSize.x - 1 - grab.x);
            InventoryDrag.GrabOffset = new Vector2Int(Mathf.Clamp(turned.x, 0, newSize.x - 1), Mathf.Clamp(turned.y, 0, newSize.y - 1));

            ShapeLifted(slot);
            MoveLifted();
            UpdatePreview();
        }

        // The held slot takes the shape of the dragged item as it is turned now.
        private void ShapeLifted(InventorySlot slot)
        {
            if (_lifted == null)
            {
                return;
            }

            _lifted.RectTransform.sizeDelta = Footprint(slot.Item.GetSize(InventoryDrag.Rotation));
            _lifted.SetIconRotation(InventoryDrag.Rotation, IconSize(slot.Item));
        }

        // The held slot goes where the pointer is, with the cell it was grabbed by under it.
        private void MoveLifted()
        {
            if (_lifted == null
                || !RectTransformUtility.ScreenPointToWorldPointInRectangle(itemLayer, _lastPointer, EventCamera, out var world))
            {
                return;
            }

            var grab = InventoryDrag.GrabOffset;
            var scale = itemLayer.lossyScale;
            _lifted.RectTransform.position = world
                                             + new Vector3(-(grab.x + 0.5f) * Pitch * scale.x, (grab.y + 0.5f) * Pitch * scale.y, 0f);
        }

        // A released item glides to its place. Without DOTween (or with a duration of 0) it jumps.
        private void Settle(InventorySlotButton item, Vector2 position, Vector2 size)
        {
            var rect = item.RectTransform;
            rect.sizeDelta = size;

#if HAS_DOTWEEN
            if (settleDuration > 0f && item.gameObject.activeInHierarchy)
            {
                DOTween.Kill(rect);
                DOTween.To(() => rect.anchoredPosition, value => rect.anchoredPosition = value, position, settleDuration)
                    .SetEase((Ease)settleEase)
                    .SetUpdate(true)
                    .SetTarget(rect)
                    .SetLink(item.gameObject)
                    .OnKill(() => item.SetLifted(false));
                return;
            }
#endif
            rect.anchoredPosition = position;
            item.SetLifted(false);
        }

        // Stops a glide that is still under way, and puts the slot back in its place among the others.
        private static void EndFlight(InventorySlotButton item)
        {
#if HAS_DOTWEEN
            DOTween.Kill(item.RectTransform);
#endif
            item.SetLifted(false);
        }

        private void UpdatePreview()
        {
            var grid = Grid;
            if (grid == null || !TryGetSlot(InventoryDrag.SlotIndex, out var slot)
                || !RectTransformUtility.RectangleContainsScreenPoint(itemLayer, _lastPointer, EventCamera))
            {
                HidePreview();
                return;
            }

            var cell = PointerCell(_lastPointer);
            var under = grid.GetSlotIndexAt(cell);
            if (under >= 0 && under != InventoryDrag.SlotIndex && grid.Slots[under].Item == slot.Item)
            {
                var target = grid.Slots[under];
                ShowPreview(target.Position, target.Size, target.SpaceLeft > 0 ? validColor : invalidColor);
                return;
            }

            var position = cell - InventoryDrag.GrabOffset;
            var fits = grid.CanMoveTo(InventoryDrag.SlotIndex, position, InventoryDrag.Rotation);
            ShowPreview(position, slot.Item.GetSize(InventoryDrag.Rotation), fits ? validColor : invalidColor);
        }

        // One tinted rectangle over the cells the item would cover, cut to the part inside the grid.
        private void ShowPreview(Vector2Int position, Vector2Int size, Color color)
        {
            var grid = Grid;
            var min = Vector2Int.Max(position, Vector2Int.zero);
            var max = Vector2Int.Min(position + size, new Vector2Int(grid.Columns, grid.Rows));
            if (max.x <= min.x || max.y <= min.y)
            {
                HidePreview();
                return;
            }

            if (_preview == null)
            {
                _preview = Instantiate(cellTemplate, previewLayer);
                _preview.name = "Drop Preview";
                _preview.raycastTarget = false;
                PrepareCellRect(_preview.rectTransform);
            }

            _preview.color = color;
            _preview.rectTransform.anchoredPosition = CellPosition(min);
            _preview.rectTransform.sizeDelta = Footprint(max - min);
            if (!_preview.gameObject.activeSelf)
            {
                _preview.gameObject.SetActive(true);
            }
        }

        private void HidePreview()
        {
            if (_preview != null && _preview.gameObject.activeSelf)
            {
                _preview.gameObject.SetActive(false);
            }
        }

        #endregion

        #region Geometry

        private Vector2 CellPosition(Vector2Int cell) => new(cell.x * Pitch, -cell.y * Pitch);

        private Vector2 Footprint(Vector2Int cells) => new(cells.x * Pitch - cellSpacing, cells.y * Pitch - cellSpacing);

        // The icon's unturned size: turning the icon (not the footprint) keeps the sprite's proportions.
        private Vector2 IconSize(ItemDefinition item) =>
            Vector2.Max(Footprint(item.Size) - Vector2.one * (iconPadding * 2f), Vector2.one);

        private Vector2Int PointerCell(Vector2 screenPosition)
        {
            RectTransformUtility.ScreenPointToLocalPointInRectangle(itemLayer, screenPosition, EventCamera, out var local);
            return new Vector2Int(Mathf.FloorToInt(local.x / Pitch), Mathf.FloorToInt(-local.y / Pitch));
        }

        private static void PrepareLayer(RectTransform layer, Vector2 size)
        {
            layer.anchorMin = layer.anchorMax = layer.pivot = new Vector2(0f, 1f);
            layer.anchoredPosition = Vector2.zero;
            layer.sizeDelta = size;
        }

        // Takes over the cells already under Cell Layer (e.g. placed there in the editor), once.
        private void CollectCells()
        {
            if (_cellsCollected)
            {
                return;
            }

            _cellsCollected = true;
            _cells.Clear();
            for (var i = 0; i < cellLayer.childCount; i++)
            {
                if (cellLayer.GetChild(i).TryGetComponent<Image>(out var cell))
                {
                    cell.raycastTarget = false;
                    _cells.Add(cell);
                }
            }
        }

        private void PrepareCellRect(RectTransform rect)
        {
            rect.anchorMin = rect.anchorMax = rect.pivot = new Vector2(0f, 1f);
            rect.sizeDelta = new Vector2(cellSize, cellSize);
            rect.localEulerAngles = Vector3.zero;
            rect.gameObject.SetActive(true);
        }

        #endregion

        private void ResolveRotateAction()
        {
            if (_rotate != null)
            {
                return;
            }

            if (rotateAction != null && rotateAction.action != null)
            {
                _rotate = rotateAction.action;
                return;
            }

            _rotate = InputSystem.actions != null ? InputSystem.actions.FindAction("RotateItem") : null;
            if (_rotate == null)
            {
                _rotate = new InputAction("Rotate Item", InputActionType.Button, "<Keyboard>/r");
                _ownsRotate = true;
            }
        }
    }
}
