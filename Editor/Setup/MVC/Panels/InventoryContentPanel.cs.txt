#if HAS_INVENTORY_SYSTEM && HAS_UNIMVC && HAS_EVENT_SYSTEM && HAS_INPUT_SYSTEM
using System;
#if HAS_BASIC_TOOLTIP
using BasicTooltip;
#endif
using TMPro;
using UniMVC;
using UnityEngine;
using UnityEngine.EventSystems;
using UnityEngine.InputSystem;
using UnityEngine.UI;
using Object = UnityEngine.Object;

namespace InventorySystem.UI
{
    /// <summary>
    /// Base of the panels that show an inventory's slots (<see cref="InventoryListPanel"/>,
    /// <see cref="InventoryGridPanel"/>): binds to one <see cref="Inventory"/> through its C# events and
    /// redraws only the slots that change - never polls. Shared behaviour lives here: tooltips on hover (with
    /// Basic Tooltip), click events, and drag and drop within a panel, between panels (<see cref="Inventory.TransferTo"/>,
    /// e.g. player &lt;-&gt; chest) and out of the UI (<see cref="ItemDroppedOutside"/>).
    /// </summary>
    /// <remarks>
    /// A panel only listens while it is visible, and redraws fully once when it is shown again. Left-drag
    /// moves the whole stack, right-drag half of it. With a gamepad, Submit on a slot picks its stack up, the d-pad
    /// or stick moves it (to another slot, or over the grid's cells), and Submit puts it down; the "PickHalf" action
    /// (West button) picks up half, Cancel puts it back. Moving the selection over a slot shows its details.
    /// </remarks>
    public abstract class InventoryContentPanel : PanelViewBase, IDropHandler
    {
        [Tooltip("Items can be dragged: moved, merged, split (right drag), swapped and carried to another window.")]
        [SerializeField] private bool canDrag = true;

        private Inventory _subscribed;
        private Canvas _rootCanvas;
        private InputAction _pickHalf;
        private bool _ownsPickHalf;
        private bool _receiving;

        /// <summary>The inventory shown, or null.</summary>
        public Inventory Inventory { get; private set; }

        /// <summary>A dragged item was released over nothing of the UI (panel, slot index, amount dragged). Nothing is removed by the panel itself.</summary>
        public event Action<InventoryContentPanel, int, int> ItemDroppedOutside;

        /// <summary>A slot holding an item was clicked (panel, slot index, button) - e.g. to use or equip it.</summary>
        public event Action<InventoryContentPanel, int, PointerEventData.InputButton> SlotClicked;

        /// <summary>An item started being dragged out of a slot (panel, slot index) - once, as the drag begins.</summary>
        public event Action<InventoryContentPanel, int> SlotDragStarted;

        /// <summary>Whether this panel can show an inventory of <paramref name="type"/>.</summary>
        public abstract bool Supports(InventoryType type);

        protected Canvas RootCanvas
        {
            get
            {
                if (_rootCanvas == null)
                {
                    var canvas = GetComponentInParent<Canvas>(true);
                    _rootCanvas = canvas != null ? canvas.rootCanvas : null;
                }

                return _rootCanvas;
            }
        }

        /// <summary>The camera screen points are converted with: null for Screen Space - Overlay canvases.</summary>
        protected Camera EventCamera =>
            RootCanvas == null || RootCanvas.renderMode == RenderMode.ScreenSpaceOverlay ? null : RootCanvas.worldCamera;

        /// <summary>Shows <paramref name="inventory"/> (null to show nothing).</summary>
        public void Bind(Inventory inventory)
        {
            if (ReferenceEquals(inventory, Inventory))
            {
                return;
            }

            CancelDrag();
            Unsubscribe();
            Inventory = inventory;

            if (isActiveAndEnabled)
            {
                Subscribe();
                Rebuild();
            }
        }

        protected virtual void OnEnable()
        {
            // The project's "PickHalf" action, or West on a gamepad: picks up half of the selected stack.
            _pickHalf = InputSystem.actions != null ? InputSystem.actions.FindAction("PickHalf") : null;
            if (_pickHalf == null)
            {
                _pickHalf = new InputAction("Pick Half", InputActionType.Button, "<Gamepad>/buttonWest");
                _ownsPickHalf = true;
            }

            _pickHalf.performed += OnPickHalfPerformed;
            _pickHalf.Enable();

            InventoryDrag.PadHoldChanged += OnPadHoldChanged;
            OnPadHoldChanged(InventoryDrag.IsActive && InventoryDrag.ViaPad);

            Subscribe();
            Rebuild();
        }

        protected virtual void OnDisable()
        {
            _pickHalf.performed -= OnPickHalfPerformed;
            if (_ownsPickHalf)
            {
                _pickHalf.Disable();
                _pickHalf.Dispose();
                _ownsPickHalf = false;
            }

            _pickHalf = null;
            InventoryDrag.PadHoldChanged -= OnPadHoldChanged;

            Unsubscribe();
            CancelDrag();
            HideTooltip();
        }

        #region Drawing

        /// <summary>Redraws everything. Called when the panel is shown or its inventory changes.</summary>
        protected abstract void Rebuild();

        /// <summary>Redraws one slot.</summary>
        protected abstract void RefreshSlot(int index);

        /// <summary>The inventory gained or lost slots.</summary>
        protected abstract void OnSlotCountChanged(int count);

        /// <summary>Creates a slot button from <paramref name="template"/> under <paramref name="parent"/>, wired to this panel.</summary>
        protected InventorySlotButton CreateSlot(InventorySlotButton template, Transform parent, int slotIndex)
        {
            var slot = Instantiate(template, parent);
            slot.name = $"Slot {slotIndex}";
            slot.Owner = this;
            slot.SlotIndex = slotIndex;
            slot.Initialize(UI);

            // The template may have navigation switched off (it was for the mouse only); the d-pad needs it.
            slot.SetNavigable(true);
            slot.SetReceiving(_receiving);
            return slot;
        }

        private void Subscribe()
        {
            if (Inventory == null || _subscribed != null)
            {
                return;
            }

            Inventory.SlotChanged += HandleSlotChanged;
            Inventory.SlotCountChanged += OnSlotCountChanged;
            Inventory.LayoutChanged += HandleLayoutChanged;
            _subscribed = Inventory;
        }

        private void Unsubscribe()
        {
            if (_subscribed == null)
            {
                return;
            }

            _subscribed.SlotChanged -= HandleSlotChanged;
            _subscribed.SlotCountChanged -= OnSlotCountChanged;
            _subscribed.LayoutChanged -= HandleLayoutChanged;
            _subscribed = null;
        }

        private void HandleSlotChanged(int index)
        {
            RefreshSlot(index);

            // The dragged stack was used up by something else (code, another panel) mid-drag.
            if (InventoryDrag.SourcePanel == this && InventoryDrag.SlotIndex == index && Inventory.Slots[index].IsEmpty)
            {
                CancelDrag();
            }
        }

        // The whole layout changed (e.g. a grid resized): a drag in progress no longer matches it.
        private void HandleLayoutChanged()
        {
            CancelDrag();
            Rebuild();
        }

        #endregion

        #region Pointer events forwarded by InventorySlotButton

        internal void OnSlotClicked(InventorySlotButton slot, PointerEventData.InputButton button)
        {
            if (button == PointerEventData.InputButton.Left)
            {
                if (InventoryDrag.IsActive && InventoryDrag.ViaPad)
                {
                    // Submit on a slot while carrying: put the item down there. A mouse click instead forgets the carried item.
                    if (UIInput.IsGamepadActive)
                    {
                        PutDown(slot);
                        return;
                    }

                    InventoryDrag.SourcePanel.CancelDrag();
                }
                else if (UIInput.IsGamepadActive && !InventoryDrag.IsActive && TryGetSlot(slot.SlotIndex, out _))
                {
                    // Submit with a gamepad picks the stack up instead of just clicking it.
                    PickUp(slot, half: false);
                    return;
                }
            }

            if (TryGetSlot(slot.SlotIndex, out _))
            {
                SlotClicked?.Invoke(this, slot.SlotIndex, button);
            }
        }

        // The selection (d-pad / stick) reached a slot: a carried item follows it, otherwise its details show.
        internal void OnSlotFocused(InventorySlotButton slot)
        {
            if (!UIInput.IsGamepadActive)
            {
                return;
            }

            if (InventoryDrag.IsActive && InventoryDrag.ViaPad)
            {
                if (RootCanvas != null)
                {
                    DragGhost.For(RootCanvas).MoveTo(SlotScreenPosition(slot));
                }

                return;
            }

            if (TryGetSlot(slot.SlotIndex, out _))
            {
                SlotClicked?.Invoke(this, slot.SlotIndex, PointerEventData.InputButton.Left);
            }
        }

        internal void OnSlotBeginDrag(InventorySlotButton slot, PointerEventData eventData)
        {
            var draggable = canDrag
                            && (eventData.button == PointerEventData.InputButton.Left || eventData.button == PointerEventData.InputButton.Right);
            if (!draggable || InventoryDrag.IsActive || RootCanvas == null || !TryGetSlot(slot.SlotIndex, out var inventorySlot))
            {
                // Cancels the drag: uGUI sends no OnDrag / OnEndDrag / OnDrop without a pointerDrag.
                eventData.pointerDrag = null;
                return;
            }

            var amount = eventData.button == PointerEventData.InputButton.Right && inventorySlot.Amount > 1
                ? (inventorySlot.Amount + 1) / 2
                : inventorySlot.Amount;

            BeginHold(slot, inventorySlot, amount, eventData, viaPad: false);
        }

        private void BeginHold(InventorySlotButton slot, InventorySlot inventorySlot, int amount, PointerEventData eventData, bool viaPad)
        {
            HideTooltip();
            InventoryDrag.Begin(this, slot, slot.SlotIndex, amount, inventorySlot.Rotation, viaPad);
            OnDragStarted(slot, inventorySlot, eventData);
            DragGhost.For(RootCanvas).MoveTo(eventData.position);
            SlotDragStarted?.Invoke(this, slot.SlotIndex);
        }

        internal void OnSlotDrag(PointerEventData eventData)
        {
            if (InventoryDrag.SourcePanel != this)
            {
                return;
            }

            DragGhost.For(RootCanvas).MoveTo(eventData.position);
            OnDragMoved(eventData);
        }

        internal void OnSlotEndDrag(PointerEventData eventData)
        {
            if (InventoryDrag.SourcePanel != this)
            {
                return;
            }

            // OnDrop (on whatever was under the pointer) has already run by now.
            var outside = !InventoryDrag.Handled && eventData.pointerCurrentRaycast.gameObject == null;
            var slotIndex = InventoryDrag.SlotIndex;
            var amount = InventoryDrag.Amount;

            CancelDrag();

            if (outside)
            {
                ItemDroppedOutside?.Invoke(this, slotIndex, amount);
            }
        }

        internal void OnSlotDrop(InventorySlotButton target, PointerEventData eventData) => HandleDrop(target, eventData);

        /// <summary>A drop on the panel itself, not on a slot (e.g. an empty grid cell).</summary>
        public void OnDrop(PointerEventData eventData) => HandleDrop(null, eventData);

        #endregion

        #region Gamepad: pick up and put down

        // Picks a stack (or half of it) up as a drag would, with a pointer standing on the slot, so everything built
        // for the mouse - the ghost, the grid's lifted item and drop preview, rotating - works the same.
        private void PickUp(InventorySlotButton slot, bool half)
        {
            if (!canDrag || RootCanvas == null || InventoryDrag.IsActive || !TryGetSlot(slot.SlotIndex, out var inventorySlot))
            {
                return;
            }

            var amount = half && inventorySlot.Amount > 1 ? (inventorySlot.Amount + 1) / 2 : inventorySlot.Amount;
            BeginHold(slot, inventorySlot, amount, PadPointer(SlotScreenPosition(slot)), viaPad: true);
        }

        // Drops the carried item where Submit was pressed - this panel's slot, or the grid cell the cursor is on.
        private void PutDown(InventorySlotButton target)
        {
            var source = InventoryDrag.SourcePanel;
            HandleDrop(target, PadDropPointer(target));
            source.CancelDrag();
        }

        private void OnPickHalfPerformed(InputAction.CallbackContext context)
        {
            if (this == null || InventoryDrag.IsActive || !UIInput.IsGamepadActive)
            {
                return;
            }

            var eventSystem = UnityEngine.EventSystems.EventSystem.current;
            var selected = eventSystem != null ? eventSystem.currentSelectedGameObject : null;
            if (selected != null && selected.TryGetComponent<InventorySlotButton>(out var slot) && slot.Owner == this)
            {
                PickUp(slot, half: true);
            }
        }

        private void OnPadHoldChanged(bool holding)
        {
            _receiving = holding;
            SetReceiving(holding);
        }

        /// <summary>While a gamepad player carries an item, every slot of the panel can be selected - empty ones too.</summary>
        protected virtual void SetReceiving(bool receiving)
        {
        }

        /// <summary>The pointer a drop with the gamepad stands for. By default: on the slot that was pressed.</summary>
        protected virtual PointerEventData PadDropPointer(InventorySlotButton target) => PadPointer(SlotScreenPosition(target));

        protected static PointerEventData PadPointer(Vector2 screenPosition) =>
            new(UnityEngine.EventSystems.EventSystem.current) { button = PointerEventData.InputButton.Left, position = screenPosition };

        /// <summary>The middle of <paramref name="slot"/> on screen.</summary>
        protected Vector2 SlotScreenPosition(InventorySlotButton slot)
        {
            var rect = slot.RectTransform;
            return RectTransformUtility.WorldToScreenPoint(EventCamera, rect.TransformPoint(rect.rect.center));
        }

        /// <summary>Puts the carried item back (for Cancel).</summary>
        internal void CancelHold() => CancelDrag();

        #endregion

        #region Drag and drop

        /// <summary>Sets up the drag ghost. By default it copies the dragged slot's size.</summary>
        protected virtual void OnDragStarted(InventorySlotButton slot, InventorySlot inventorySlot, PointerEventData eventData)
        {
            var size = slot.RectTransform.rect.size;
            DragGhost.For(RootCanvas).Show(inventorySlot.Item.Icon, InventoryDrag.Amount, size, new Vector2(0.5f, 0.5f), 0, size * 0.8f);
        }

        /// <summary>The pointer moved while this panel's item is dragged.</summary>
        protected virtual void OnDragMoved(PointerEventData eventData)
        {
        }

        /// <summary>This panel's drag ended, dropped or not.</summary>
        protected virtual void OnDragEnded()
        {
        }

        /// <summary>
        /// A drag from the same inventory was dropped on this panel; <paramref name="target"/> is the slot
        /// under the pointer, or null. By default: <see cref="Inventory.Move"/> onto that slot.
        /// </summary>
        protected virtual void DropFromSameInventory(InventorySlotButton target, PointerEventData eventData)
        {
            if (target != null && target.SlotIndex != InventoryDrag.SlotIndex)
            {
                Inventory.Move(InventoryDrag.SlotIndex, target.SlotIndex, InventoryDrag.Amount);
            }
        }

        /// <summary>Stops this panel's drag without dropping anything.</summary>
        protected void CancelDrag()
        {
            if (InventoryDrag.SourcePanel != this)
            {
                return;
            }

            InventoryDrag.End();
            OnDragEnded();
        }

        private void HandleDrop(InventorySlotButton target, PointerEventData eventData)
        {
            var source = InventoryDrag.SourcePanel;
            if (source == null || source.Inventory == null || Inventory == null)
            {
                return;
            }

            InventoryDrag.Handled = true;

            if (ReferenceEquals(source.Inventory, Inventory))
            {
                DropFromSameInventory(target, eventData);
            }
            else
            {
                source.Inventory.TransferTo(Inventory, InventoryDrag.SlotIndex, InventoryDrag.Amount);
            }
        }

        #endregion

        /// <summary>The non-empty slot at <paramref name="index"/>, if there is one.</summary>
        protected bool TryGetSlot(int index, out InventorySlot slot)
        {
            slot = Inventory != null && (uint)index < (uint)Inventory.SlotCount ? Inventory.Slots[index] : null;
            return slot != null && !slot.IsEmpty;
        }

        // Only a tooltip shown for one of this panel's slots.
        private void HideTooltip()
        {
#if HAS_BASIC_TOOLTIP
            var target = Tooltips.Target;
            if (target != null && target.IsChildOf(transform))
            {
                Tooltips.HideAny();
            }
#endif
        }

#if HAS_BASIC_TOOLTIP
        /// <summary>
        /// What <paramref name="slot"/>'s tooltip shows: its item's name, description (both translated with
        /// LocalizationSystem) and icon. None for an empty slot, or while an item is dragged. Asked on hover only,
        /// never per frame.
        /// </summary>
        internal bool TryGetTooltip(InventorySlotButton slot, out TooltipContent content)
        {
            if (InventoryDrag.IsActive || !TryGetSlot(slot.SlotIndex, out var inventorySlot))
            {
                content = default;
                return false;
            }

            var item = inventorySlot.Item;
            var description = string.IsNullOrEmpty(item.Description) ? null : Localized(item.Description);
            content = new TooltipContent(Localized(item.DisplayName), description, item.Icon);
            return true;
        }

        private static string Localized(string text) =>
#if HAS_LOCALIZATION_SYSTEM
            LocalizationSystem.LocalizationRuntime.Get(text);
#else
            text;
#endif
#endif
    }

    /// <summary>The one drag in progress, shared by every panel so an item can be dragged from one to another.</summary>
    internal static class InventoryDrag
    {
        public static InventoryContentPanel SourcePanel { get; private set; }
        public static InventorySlotButton SourceSlot { get; private set; }
        public static int SlotIndex { get; private set; }

        /// <summary>How many are dragged: the whole stack (left button) or half of it (right button).</summary>
        public static int Amount { get; private set; }

        /// <summary>Grid drags: the rotation the item will be dropped in. Changed by the rotate input.</summary>
        public static int Rotation { get; set; }

        /// <summary>Grid drags: which cell of the item the pointer holds, relative to its top-left cell.</summary>
        public static Vector2Int GrabOffset { get; set; }

        /// <summary>Set by the panel the item was dropped on, so the source knows it didn't land outside.</summary>
        public static bool Handled { get; set; }

        public static bool IsActive => SourcePanel != null;

        /// <summary>The item is carried by a gamepad player (picked up with Submit), not dragged by the pointer.</summary>
        public static bool ViaPad { get; private set; }

        /// <summary>Raised with true when a gamepad player picks an item up and with false when it is put down or cancelled.</summary>
        public static event Action<bool> PadHoldChanged;

        public static void Begin(InventoryContentPanel panel, InventorySlotButton slot, int slotIndex, int amount, int rotation, bool viaPad = false)
        {
            SourcePanel = panel;
            SourceSlot = slot;
            SlotIndex = slotIndex;
            Amount = amount;
            Rotation = rotation;
            GrabOffset = Vector2Int.zero;
            Handled = false;
            ViaPad = viaPad;
            slot.SetDragged(true);

            if (viaPad)
            {
                PadHoldChanged?.Invoke(true);
            }
        }

        public static void End()
        {
            if (SourceSlot != null)
            {
                SourceSlot.SetDragged(false);
            }

            DragGhost.HideCurrent();
            SourcePanel = null;
            SourceSlot = null;

            if (ViaPad)
            {
                ViaPad = false;
                PadHoldChanged?.Invoke(false);
            }
        }

        // Keeps the static state clean when "Enter Play Mode Options" skips the domain reload.
        [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.SubsystemRegistration)]
        private static void ResetStatics()
        {
            SourcePanel = null;
            SourceSlot = null;
            ViaPad = false;
            PadHoldChanged = null;
        }
    }

    /// <summary>The icon that follows the pointer during a drag, built on first use as the last child of the root canvas.</summary>
    internal sealed class DragGhost
    {
        private static DragGhost _current;

        private readonly Canvas _canvas;
        private readonly RectTransform _root;
        private readonly RectTransform _iconRect;
        private readonly Image _icon;
        private readonly TextMeshProUGUI _amount;

        private DragGhost(Canvas canvas)
        {
            _canvas = canvas;

            var root = new GameObject("Inventory Drag Ghost", typeof(RectTransform), typeof(CanvasGroup));
            root.layer = canvas.gameObject.layer;
            _root = (RectTransform)root.transform;
            _root.SetParent(canvas.transform, false);
            _root.anchorMin = _root.anchorMax = new Vector2(0.5f, 0.5f);

            var group = root.GetComponent<CanvasGroup>();
            group.blocksRaycasts = false;
            group.interactable = false;
            group.alpha = 0.85f;

            var icon = new GameObject("Icon", typeof(RectTransform), typeof(Image));
            icon.layer = root.layer;
            _iconRect = (RectTransform)icon.transform;
            _iconRect.SetParent(_root, false);
            _iconRect.anchorMin = _iconRect.anchorMax = new Vector2(0.5f, 0.5f);
            _icon = icon.GetComponent<Image>();
            _icon.raycastTarget = false;
            _icon.preserveAspect = true;

            var amount = new GameObject("Amount", typeof(RectTransform));
            amount.layer = root.layer;
            var amountRect = (RectTransform)amount.transform;
            amountRect.SetParent(_root, false);
            amountRect.anchorMin = Vector2.zero;
            amountRect.anchorMax = Vector2.one;
            amountRect.offsetMin = new Vector2(4f, 2f);
            amountRect.offsetMax = new Vector2(-4f, -2f);
            _amount = amount.AddComponent<TextMeshProUGUI>();
            _amount.raycastTarget = false;
            _amount.fontSize = 18f;
            _amount.fontStyle = FontStyles.Bold;
            _amount.alignment = TextAlignmentOptions.BottomRight;

            root.SetActive(false);
        }

        /// <summary>The ghost of <paramref name="canvas"/>'s root canvas, built if needed.</summary>
        public static DragGhost For(Canvas canvas)
        {
            var rootCanvas = canvas.rootCanvas;
            if (_current == null || _current._root == null || _current._canvas != rootCanvas)
            {
                if (_current != null && _current._root != null)
                {
                    Object.Destroy(_current._root.gameObject);
                }

                _current = new DragGhost(rootCanvas);
            }

            return _current;
        }

        public static void HideCurrent()
        {
            if (_current != null && _current._root != null)
            {
                _current._root.gameObject.SetActive(false);
            }
        }

        /// <param name="size">Size of the ghost, in canvas units.</param>
        /// <param name="pivot">The point of the ghost under the pointer (0-1).</param>
        /// <param name="rotation">Clockwise quarter turns of the icon.</param>
        /// <param name="iconSize">The icon's unrotated size, in canvas units.</param>
        public void Show(Sprite sprite, int amount, Vector2 size, Vector2 pivot, int rotation, Vector2 iconSize)
        {
            _root.gameObject.SetActive(true);
            _root.SetAsLastSibling();
            _root.sizeDelta = size;
            _root.pivot = pivot;

            _icon.sprite = sprite;
            _icon.enabled = sprite != null;
            _iconRect.sizeDelta = iconSize;
            _iconRect.localEulerAngles = new Vector3(0f, 0f, -90f * rotation);

            if (amount > 1)
            {
                _amount.SetText("{0}", amount);
            }
            else
            {
                _amount.SetText(string.Empty);
            }
        }

        public void MoveTo(Vector2 screenPosition)
        {
            var camera = _canvas.renderMode == RenderMode.ScreenSpaceOverlay ? null : _canvas.worldCamera;
            if (RectTransformUtility.ScreenPointToLocalPointInRectangle((RectTransform)_canvas.transform, screenPosition, camera, out var local))
            {
                _root.localPosition = local;
            }
        }
    }
