#if HAS_INVENTORY_SYSTEM && HAS_UNIMVC && HAS_EVENT_SYSTEM && HAS_INPUT_SYSTEM
using System.Collections.Generic;
using UniMVC;
using UnityEngine;

namespace InventorySystem.UI
{
    /// <summary>
    /// Pickup toasts ("+3 Apple") and "no room" warnings for the main inventory. It doesn't listen to
    /// anything itself: <see cref="InventoryController"/> calls <see cref="Notify"/>, and the panel shows
    /// itself for as long as a toast is on screen, then hides again. Picking up more of an item that still
    /// has a toast on screen adds to that toast instead of stacking a new one. Toasts are pooled; timing is
    /// unscaled, so they also fade while the game is paused.
    /// </summary>
    public sealed class InventoryNotificationPanel : PanelViewBase
    {
        [SerializeField] private InventoryToastPanel toastTemplate;

        [Tooltip("Toasts are added here as the last child; give it a VerticalLayoutGroup. Empty: this object.")]
        [SerializeField] private RectTransform container;

        [Min(1)] [SerializeField] private int maxToasts = 5;
        [Min(0.1f)] [SerializeField] private float displaySeconds = 2.5f;
        [Min(0f)] [SerializeField] private float fadeSeconds = 0.4f;

        [SerializeField] private Color addedColor = Color.white;
        [SerializeField] private Color fullColor = new(1f, 0.45f, 0.4f, 1f);

        [Tooltip("{0} = amount, {1} = item name.")]
        [SerializeField] private string addedFormat = "+{0} {1}";

        [Tooltip("{0} = amount that didn't fit, {1} = item name.")]
        [SerializeField] private string fullFormat = "Inventory full: {1}";

        private readonly List<InventoryToastPanel> _active = new();
        private readonly Stack<InventoryToastPanel> _pool = new();

        protected override void OnInitialize()
        {
            base.OnInitialize();

            // Hidden until the first toast.
            if (IsVisible)
            {
                gameObject.SetActive(false);
            }
        }

        /// <summary>Shows "+<paramref name="amount"/> item", or "Inventory full: item" when <paramref name="full"/>.</summary>
        public void Notify(ItemDefinition item, int amount, bool full)
        {
            if (item == null || toastTemplate == null)
            {
                return;
            }

            Show();

            var toast = FindActive(item, full);
            if (toast != null)
            {
                toast.Amount += amount;
            }
            else
            {
                toast = Take();
                toast.Item = item;
                toast.IsFull = full;
                toast.Amount = amount;
            }

            toast.transform.SetAsLastSibling();
            toast.Set(item.Icon, string.Format(full ? fullFormat : addedFormat, toast.Amount, Localized(item.DisplayName)),
                full ? fullColor : addedColor);
            toast.TimeLeft = displaySeconds + fadeSeconds;
            toast.Group.alpha = 1f;
        }

        private void Update()
        {
            var deltaTime = Time.unscaledDeltaTime;
            for (var i = _active.Count - 1; i >= 0; i--)
            {
                var toast = _active[i];
                toast.TimeLeft -= deltaTime;
                if (toast.TimeLeft <= 0f)
                {
                    Release(toast);
                }
                else if (toast.TimeLeft < fadeSeconds)
                {
                    toast.Group.alpha = toast.TimeLeft / fadeSeconds;
                }
            }

            if (_active.Count == 0)
            {
                Hide();
            }
        }

        private InventoryToastPanel FindActive(ItemDefinition item, bool full)
        {
            foreach (var toast in _active)
            {
                if (toast.Item == item && toast.IsFull == full)
                {
                    return toast;
                }
            }

            return null;
        }

        private InventoryToastPanel Take()
        {
            // Too many on screen: recycle the oldest.
            if (_active.Count >= maxToasts)
            {
                Release(_active[0]);
            }

            var toast = _pool.Count > 0 ? _pool.Pop() : Instantiate(toastTemplate, container != null ? container : transform);
            toast.gameObject.SetActive(true);
            _active.Add(toast);
            return toast;
        }

        private void Release(InventoryToastPanel toast)
        {
            toast.Item = null;
            toast.gameObject.SetActive(false);
            _active.Remove(toast);
            _pool.Push(toast);
        }

        private static string Localized(string text) =>
#if HAS_LOCALIZATION_SYSTEM
            LocalizationSystem.LocalizationRuntime.Get(text);
#else
            text;
#endif
    }
}
#endif
