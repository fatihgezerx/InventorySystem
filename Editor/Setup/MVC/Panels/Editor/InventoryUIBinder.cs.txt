#if HAS_INVENTORY_SYSTEM && HAS_UNIMVC && HAS_EVENT_SYSTEM && HAS_INPUT_SYSTEM && HAS_EASYUI
using System.Collections.Generic;
using System.Text;
using EasyUI;
using InventorySystem.UI;
using TMPro;
using UniMVC;
using UnityEditor;
using UnityEngine;
using UnityEngine.InputSystem.UI;
using UnityEngine.UI;
using Object = UnityEngine.Object;

namespace InventorySystem.UIEditor
{
    /// <summary>
    /// Sets up an Easy UI panel built into the scene (<c>GameObject &gt; UI (Canvas) &gt; Easy UI</c>) that has
    /// <see cref="InventoryRoles"/>: adds the inventory's views to the elements with them, wired to each other.
    /// </summary>
    /// <remarks>
    /// <list type="bullet">
    /// <item>The Inventory Panel is the window: it gets the <see cref="InventoryPanel"/>, so it opens and closes whole.</item>
    /// <item>The Slot Container's role says the content panel: an <see cref="InventoryGridPanel"/> for (Grid) - cells
    /// made right away from the InventoryData's Columns x Rows, sized by its Grid Layout Group or the Cell Template -
    /// otherwise an <see cref="InventoryListPanel"/> laid out by its Grid Layout Group.</item>
    /// <item>The roles next to the Slot Template say what the slots do: Clickable (select on click), Draggable and Drop To
    /// World. With Basic Tooltip's Shows Tooltip on it, every slot gives its item's tooltip.</item>
    /// <item>Boxes follow from what is in them: the element holding the Toast Message is the toast template, and its
    /// parent the notifications, taken out of the window to the canvas so they show while the inventory is closed.
    /// Item Name, Description, Icon and Examine View are shown by an <see cref="InventoryDetailsPanel"/> on an
    /// invisible object in the window.</item>
    /// <item>Templates are hidden; a Slot Template without an icon or amount gets one.</item>
    /// <item>The canvas gets a <see cref="UIManager"/> and an <see cref="InventoryController"/> if it has none (and the
    /// scene an EventSystem), and every view is listed where it belongs.</item>
    /// </list>
    /// It lives in your MVC folder (not in <c>MVC/Editor/</c>, UniMVC's own editor assembly, which can't see your
    /// views), and Easy UI finds it on its own.
    /// </remarks>
    internal sealed class InventoryUIBinder : IEasyUIBuildHandler
    {
        private static readonly Color Transparent = new(0f, 0f, 0f, 0f);

        // Grid cells made when there is no InventoryData to read Columns x Rows from.
        private const int FallbackColumns = 6;
        private const int FallbackRows = 4;

        private readonly List<EasyUINode> _nodes = new();

        public int Order => 0;

        public void OnBuilt(EasyUIBuild build)
        {
            var document = build.Document;
            if (!InventoryRoles.IsUsedIn(document))
            {
                return;
            }

            var windowObject = build.Find(InventoryRoles.Panel);
            var gridObject = build.Find(InventoryRoles.GridContainer);
            var container = gridObject != null ? gridObject : build.Find(InventoryRoles.ListContainer);
            var slotObject = build.Find(InventoryRoles.SlotTemplate);
            if (!HasRequired(build, windowObject, container, slotObject))
            {
                return;
            }

            if (!container.transform.IsChildOf(windowObject.transform))
            {
                Debug.LogWarning("[Inventory System] The Slot Container isn't inside the Inventory Panel, so it won't open and close with it.", container);
            }

            // What the slots do: the roles next to the Slot Template.
            var slotNode = EasyUIRoles.FindNode(document, InventoryRoles.SlotTemplate);
            var clickable = slotNode.HasRole(InventoryRoles.Clickable);
            var draggable = slotNode.HasRole(InventoryRoles.Draggable);
            var dropToWorld = draggable && slotNode.HasRole(InventoryRoles.DropToWorld);

            // Content: the slots, and for grids the cells.
            var slot = BuildSlotTemplate(slotObject, build.FindInside(slotObject, InventoryRoles.ItemIcon), build.FindInside(slotObject, InventoryRoles.SlotAmount));
            InventoryContentPanel content = gridObject != null
                ? BuildGrid(gridObject, slot, build.Find(InventoryRoles.CellTemplate))
                : BuildList(container, slot);
            new Fields(content).Bool("canDrag", draggable).Apply();

            // The window.
            if (!windowObject.TryGetComponent<InventoryPanel>(out var window))
            {
                window = windowObject.AddComponent<InventoryPanel>();
            }

            var organize = AddButton<InventoryOrganizeButton>(build.Find(InventoryRoles.OrganizeButton));
            AddButton<InventoryCloseButton>(build.Find(InventoryRoles.CloseButton));
            var details = BuildDetails(build, windowObject);
            AddButton<InventoryDropButton>(build.Find(InventoryRoles.DropButton));
            AddButton<InventoryUseButton>(build.Find(InventoryRoles.UseButton));

            // The weight bar is a Slider the player can't drag.
            var weightObject = build.Find(InventoryRoles.WeightBar);
            var weight = weightObject != null ? weightObject.GetComponent<Slider>() : null;
            if (weight != null)
            {
                weight.interactable = false;
            }

            new Fields(window)
                .Objects("contents", content)
                .Ref("capacityText", Label(build.Find(InventoryRoles.CapacityText)))
                .Ref("weightSlider", weight)
                .Ref("organizeButton", organize != null ? organize.gameObject : null)
                .Ref("details", details)
                .Bool("selectOnClick", clickable)
                .Bool("dropOutside", dropToWorld)
                .Bool("blocksGameplay", true)
                .Apply();

            var notifications = BuildNotifications(build, windowObject, out var toast);

            // Templates are copied at runtime and set up by their panels: not listed.
            var ui = UISetup.EnsureUIManager(build.Canvas);
            UISetup.EnsureController<InventoryController>(ui);
            EnsureEventSystem();
            UISetup.ListViews(ui, build.Root.transform, slot);
            if (notifications != null)
            {
                UISetup.ListViews(ui, notifications.transform, toast);
            }
        }

        // The window, a slot container and the slot template: without them there is no inventory to set up.
        private static bool HasRequired(EasyUIBuild build, GameObject window, GameObject container, GameObject slot)
        {
            var missing = new StringBuilder();
            if (window == null)
            {
                missing.Append("\n- Inventory Panel (usually on the panel's root)");
            }

            if (container == null)
            {
                missing.Append("\n- Slot Container (List) or Slot Container (Grid)");
            }

            if (slot == null)
            {
                missing.Append("\n- Slot Template");
            }

            if (missing.Length == 0)
            {
                return true;
            }

            Debug.LogError($"[Inventory System] \"{build.Panel.name}\" has inventory roles but was built without the inventory's views: " +
                           $"it has no element with{missing}\nMark them in Easy UI (Add Role > Inventory) and build it again.", build.Root);
            return false;
        }

        #region Content

        // The Slot Template becomes an InventorySlotButton, with an icon and an amount (made when the panel has none).
        private static InventorySlotButton BuildSlotTemplate(GameObject go, GameObject iconObject, GameObject amountObject)
        {
            if (!go.TryGetComponent<Button>(out var button))
            {
                button = go.AddComponent<Button>();
                button.targetGraphic = go.TryGetComponent<Graphic>(out var graphic) ? graphic : AddTransparentImage(go);
            }

            button.navigation = new Navigation { mode = Navigation.Mode.None };

            // An empty slot is a disabled button (it can't be pressed): it keeps the slot's look.
            var colors = button.colors;
            colors.disabledColor = colors.normalColor;
            button.colors = colors;

            var icon = iconObject != null ? iconObject.GetComponent<Image>() : null;
            if (icon == null)
            {
                icon = NewChild("Icon", go.transform, 8f).AddComponent<Image>();
                icon.preserveAspect = true;
            }

            icon.raycastTarget = false;

            var amount = amountObject != null ? amountObject.GetComponent<TMP_Text>() : null;
            if (amount == null)
            {
                amount = NewChild("Amount", go.transform, 5f).AddComponent<TextMeshProUGUI>();
                amount.fontSize = 18f;
                amount.fontStyle = FontStyles.Bold;
                amount.alignment = TextAlignmentOptions.BottomRight;
                amount.textWrappingMode = TextWrappingModes.NoWrap;
            }

            amount.raycastTarget = false;
            amount.text = string.Empty;

            // Designed inside the icon (in Easy UI a text can only lie over an image as its child): moved up to the
            // slot, where it was, and drawn over the icon - so it doesn't turn with a rotated grid item's icon.
            if (amount.transform.IsChildOf(icon.transform))
            {
                amount.transform.SetParent(go.transform, true);
                amount.transform.SetAsLastSibling();
            }

            var slot = go.AddComponent<InventorySlotButton>();
            new Fields(slot).Ref("icon", icon).Ref("amountText", amount).Apply();
            go.SetActive(false);
            return slot;
        }

        // Unlimited / Weight / Max Slot: slots laid out by the container's layout group, as designed. A Grid Layout
        // Group: rows of as many columns as fit, with as many rows visible as fit. A Horizontal / Vertical one: a
        // single row / column. Without one, a Grid Layout Group at the Slot Template's size. Placeholder cells fill
        // what is seen - the Scroll View's viewport, or the container itself.
        private static InventoryListPanel BuildList(GameObject go, InventorySlotButton slot)
        {
            var slotSize = ((RectTransform)slot.transform).rect.size;
            var layout = go.GetComponent<LayoutGroup>();
            if (layout == null)
            {
                var made = go.AddComponent<GridLayoutGroup>();
                made.cellSize = slotSize;
                made.spacing = new Vector2(6f, 6f);
                layout = made;
            }

            var rect = (RectTransform)go.transform;
            var scroll = go.GetComponentInParent<ScrollRect>(true);
            var scrolled = scroll != null && scroll.content == rect;
            var view = scrolled && scroll.viewport != null ? scroll.viewport.rect.size : rect.rect.size;

            int columns, rows, minimum;
            var fitHorizontally = false;
            if (layout is GridLayoutGroup grid)
            {
                var pitch = grid.cellSize + grid.spacing;
                columns = grid.constraint == GridLayoutGroup.Constraint.FixedColumnCount
                    ? grid.constraintCount
                    : Mathf.Max(1, Mathf.FloorToInt((view.x - grid.padding.horizontal + grid.spacing.x) / Mathf.Max(pitch.x, 1f)));
                rows = Mathf.Max(1, Mathf.FloorToInt((view.y - grid.padding.vertical + grid.spacing.y) / Mathf.Max(pitch.y, 1f)));
                minimum = columns * rows;
            }
            else
            {
                var line = (HorizontalOrVerticalLayoutGroup)layout;
                fitHorizontally = line is HorizontalLayoutGroup;
                var length = fitHorizontally ? view.x - line.padding.horizontal : view.y - line.padding.vertical;
                var step = (fitHorizontally ? slotSize.x : slotSize.y) + line.spacing;
                columns = 1;
                rows = 1;
                minimum = Mathf.Max(1, Mathf.FloorToInt((length + line.spacing) / Mathf.Max(step, 1f)));
            }

            // In a Scroll View, the Content grows with its slots - along its row or column - so the view can scroll them.
            if (scrolled && !go.TryGetComponent<ContentSizeFitter>(out _))
            {
                var fitter = go.AddComponent<ContentSizeFitter>();
                if (fitHorizontally)
                {
                    fitter.horizontalFit = ContentSizeFitter.FitMode.PreferredSize;
                }
                else
                {
                    fitter.verticalFit = ContentSizeFitter.FitMode.PreferredSize;
                }
            }

            EnsureDropTarget(go);
            var panel = go.AddComponent<InventoryListPanel>();
            new Fields(panel)
                .Ref("slotTemplate", slot)
                .Ref("content", layout)
                .Ref("layoutElement", go.GetComponent<LayoutElement>())
                .Int("columns", columns)
                .Int("maxVisibleRows", rows)
                .Int("minimumVisibleSlots", minimum)
                .Apply();
            return panel;
        }

        // Grid: the panel places cells and items itself, so a Grid Layout Group on the container only gives the cell
        // size and spacing, and goes. The cells are made now, from the InventoryData's Columns x Rows, to be seen and
        // styled in the scene; the panel lays them out again for the inventory it shows.
        private static InventoryGridPanel BuildGrid(GameObject go, InventorySlotButton item, GameObject cellObject)
        {
            if (cellObject == null)
            {
                cellObject = CellFromSlot(go, item);
            }

            var cell = cellObject.GetComponent<Image>();
            var cellSize = ((RectTransform)cellObject.transform).rect.width;
            var spacing = 2f;
            if (go.TryGetComponent<LayoutGroup>(out var layout))
            {
                if (layout is GridLayoutGroup gridLayout)
                {
                    cellSize = gridLayout.cellSize.x;
                    spacing = gridLayout.spacing.x;
                }
                else if (layout is HorizontalOrVerticalLayoutGroup line)
                {
                    spacing = line.spacing;
                }

                Object.DestroyImmediate(layout);
            }

            var cells = NewChild("Cells", go.transform, 0f);
            var items = NewChild("Items", go.transform, 0f);
            var preview = NewChild("Preview", go.transform, 0f);
            cellObject.SetActive(false);

            EnsureDropTarget(go);

            // The panel sets this Layout Element's preferred size to the whole grid: a Content Size Fitter on the
            // container (or a layout group around it) then sizes it to the grid - with the layout group gone there
            // is nothing else to measure, and it would shrink to nothing, missing the drops on empty cells.
            if (!go.TryGetComponent<LayoutElement>(out var gridSize))
            {
                gridSize = go.AddComponent<LayoutElement>();
            }

            var panel = go.AddComponent<InventoryGridPanel>();
            new Fields(panel)
                .Ref("cellLayer", cells.transform)
                .Ref("itemLayer", items.transform)
                .Ref("previewLayer", preview.transform)
                .Ref("cellTemplate", cell)
                .Ref("itemTemplate", item)
                .Ref("layoutElement", gridSize)
                .Float("cellSize", Mathf.Max(8f, cellSize))
                .Float("cellSpacing", Mathf.Max(0f, spacing))
                .Apply();

            GridSize(out var columns, out var rows);
            panel.LayoutGrid(columns, rows);
            for (var i = 0; i < cells.transform.childCount; i++)
            {
                cells.transform.GetChild(i).name = $"Cell ({i % columns}, {i / columns})";
            }

            return panel;
        }

        // Columns x Rows of the project's (first) InventoryData.
        private static void GridSize(out int columns, out int rows)
        {
            foreach (var guid in AssetDatabase.FindAssets("t:" + nameof(InventoryData)))
            {
                var data = AssetDatabase.LoadAssetAtPath<InventoryData>(AssetDatabase.GUIDToAssetPath(guid));
                if (data != null)
                {
                    columns = Mathf.Max(1, data.Settings.Columns);
                    rows = Mathf.Max(1, data.Settings.Rows);
                    return;
                }
            }

            columns = FallbackColumns;
            rows = FallbackRows;
            Debug.LogWarning($"[Inventory System] No InventoryData in the project: the grid shows {columns} x {rows} cells until an inventory is shown.");
        }

        // No Cell Template: one made from the Slot Template's look - its own Image, without its icon or amount - at
        // its size (a faint square when the slot has no visible Image).
        private static GameObject CellFromSlot(GameObject container, InventorySlotButton slot)
        {
            var cell = new GameObject("Cell Template", typeof(RectTransform), typeof(Image)) { layer = container.layer };
            var rect = (RectTransform)cell.transform;
            rect.SetParent(container.transform, false);
            rect.sizeDelta = ((RectTransform)slot.transform).rect.size;

            var image = cell.GetComponent<Image>();
            image.raycastTarget = false;
            if (slot.TryGetComponent<Image>(out var look) && look.color.a > 0.01f)
            {
                image.sprite = look.sprite;
                image.type = look.type;
                image.color = look.color;
                image.pixelsPerUnitMultiplier = look.pixelsPerUnitMultiplier;
            }
            else
            {
                image.color = new Color(1f, 1f, 1f, 0.1f);
            }

            return cell;
        }

        #endregion

        #region Details, notifications

        // On an invisible object in the window, when anything shows the selected item (Item Name, Description,
        // Examine View, an Item Icon that isn't a slot's or a notification's) - wherever those are in the panel.
        private InventoryDetailsPanel BuildDetails(EasyUIBuild build, GameObject window)
        {
            var icon = SelectedItemIcon(build);
            var nameObject = build.Find(InventoryRoles.DetailsName);
            var descriptionObject = build.Find(InventoryRoles.DetailsDescription);
            var examineImage = build.Find(InventoryRoles.ExamineView);
            if (icon == null && nameObject == null && descriptionObject == null && examineImage == null)
            {
                return null;
            }

            var panel = NewChild("Details", window.transform, 0f).AddComponent<InventoryDetailsPanel>();
            var examine = examineImage != null ? examineImage.AddComponent<InventoryExamineView>() : null;

            new Fields(panel)
                .Ref("nameText", Label(nameObject))
                .Ref("descriptionText", Label(descriptionObject))
                .Ref("icon", icon != null ? icon.GetComponent<Image>() : null)
                .Ref("examine", examine)
                .Apply();
            return panel;
        }

        // The Item Icon that shows the selected item (not a slot's, nor a notification's), or null.
        private GameObject SelectedItemIcon(EasyUIBuild build)
        {
            EasyUIRoles.FindNodes(build.Document, InventoryRoles.ItemIcon, _nodes);
            foreach (var node in _nodes)
            {
                if (InventoryRoles.IsSelectedItemIcon(build.Document, node))
                {
                    var go = build.Get(node);
                    if (go != null)
                    {
                        return go;
                    }
                }
            }

            return null;
        }

        // The element holding the Toast Message is the toast template, copied for every notification (an Item Icon
        // in it shows the item); its parent stacks them. That parent is taken out of the window to the canvas, so
        // notifications show while the inventory is closed. Returns it, and the toast template.
        private static GameObject BuildNotifications(EasyUIBuild build, GameObject window, out InventoryToastPanel toast)
        {
            toast = null;
            var messageObject = build.Find(InventoryRoles.ToastMessage);
            if (messageObject == null)
            {
                return null;
            }

            var template = messageObject.transform.parent.gameObject;
            if (template == window || template == build.Root)
            {
                Debug.LogWarning("[Inventory System] The Toast Message is straight in the window: put it (and an Item Icon, if you " +
                                 "like) in a box of their own, inside another box that stacks them. No notifications were made.", messageObject);
                return null;
            }

            var iconObject = build.FindInside(template, InventoryRoles.ItemIcon);
            var container = template.transform.parent;
            if (container.gameObject == window || container.gameObject == build.Root)
            {
                // The toast sits straight in the window: a container of its own, over the same area.
                var own = NewChild(build.Root.name + " Notifications", container, 0f);
                template.transform.SetParent(own.transform, true);
                container = own.transform;
            }

            if (build.Canvas != null)
            {
                build.MoveOut(container.gameObject, build.Canvas.transform);
            }

            if (!container.TryGetComponent<LayoutGroup>(out _))
            {
                var stack = container.gameObject.AddComponent<VerticalLayoutGroup>();
                stack.spacing = 6f;
                stack.childAlignment = TextAnchor.LowerLeft;
                stack.childControlWidth = stack.childControlHeight = false;
                stack.childForceExpandWidth = stack.childForceExpandHeight = false;
            }

            var notifications = container.gameObject.AddComponent<InventoryNotificationPanel>();
            toast = template.AddComponent<InventoryToastPanel>();
            toast.GetComponent<CanvasGroup>().blocksRaycasts = false;

            var icon = iconObject != null ? iconObject.GetComponent<Image>() : null;
            if (icon == null)
            {
                icon = NewChild("Icon", template.transform, 6f).AddComponent<Image>();
                icon.preserveAspect = true;
                icon.enabled = false;
            }

            new Fields(toast).Ref("icon", icon).Ref("message", Label(messageObject)).Apply();
            new Fields(notifications)
                .Ref("toastTemplate", toast)
                .Ref("container", container)
                .Apply();
            template.SetActive(false);
            return container.gameObject;
        }

        #endregion

        #region Helpers

        // The inventory reads its input through the Input System's UI module.
        private static void EnsureEventSystem()
        {
            if (Object.FindFirstObjectByType<UnityEngine.EventSystems.EventSystem>() != null)
            {
                return;
            }

            var eventSystem = new GameObject("EventSystem", typeof(UnityEngine.EventSystems.EventSystem), typeof(InputSystemUIInputModule));
            Undo.RegisterCreatedObjectUndo(eventSystem, "Create EventSystem");
        }

        private static T AddButton<T>(GameObject go) where T : ButtonViewBase
        {
            if (go == null)
            {
                return null;
            }

            if (!go.TryGetComponent<Button>(out _))
            {
                Debug.LogWarning($"[Inventory System] '{go.name}' has no Button, so it can't be a {typeof(T).Name}.", go);
                return null;
            }

            return go.AddComponent<T>();
        }

        private static TMP_Text Label(GameObject go)
        {
            if (go == null || !go.TryGetComponent<TMP_Text>(out var text))
            {
                return null;
            }

            text.raycastTarget = false;
            return text;
        }

        private static Image AddTransparentImage(GameObject go)
        {
            var image = go.AddComponent<Image>();
            image.color = Transparent;
            return image;
        }

        // Drops between slots (or on empty grid cells) land on the content panel itself, so it needs a raycast target.
        private static void EnsureDropTarget(GameObject go)
        {
            if (!go.TryGetComponent<Graphic>(out _))
            {
                AddTransparentImage(go);
            }
        }

        // A child stretched over its parent, `inset` in from every edge.
        private static GameObject NewChild(string name, Transform parent, float inset)
        {
            var go = new GameObject(name, typeof(RectTransform)) { layer = parent.gameObject.layer };
            var rect = (RectTransform)go.transform;
            rect.SetParent(parent, false);
            rect.anchorMin = Vector2.zero;
            rect.anchorMax = Vector2.one;
            rect.offsetMin = new Vector2(inset, inset);
            rect.offsetMax = new Vector2(-inset, -inset);
            return go;
        }

        // Sets a component's serialized fields, then applies them once.
        private readonly struct Fields
        {
            private readonly SerializedObject _object;

            public Fields(Object target) => _object = new SerializedObject(target);

            public Fields Ref(string field, Object value)
            {
                _object.FindProperty(field).objectReferenceValue = value;
                return this;
            }

            public Fields Objects(string field, params Object[] values)
            {
                var array = _object.FindProperty(field);
                array.arraySize = values.Length;
                for (var i = 0; i < values.Length; i++)
                {
                    array.GetArrayElementAtIndex(i).objectReferenceValue = values[i];
                }

                return this;
            }

            public Fields Bool(string field, bool value)
            {
                _object.FindProperty(field).boolValue = value;
                return this;
            }

            public Fields Int(string field, int value)
            {
                _object.FindProperty(field).intValue = value;
                return this;
            }

            public Fields Float(string field, float value)
            {
                _object.FindProperty(field).floatValue = value;
                return this;
            }

            public void Apply() => _object.ApplyModifiedPropertiesWithoutUndo();
        }

        #endregion
    }
