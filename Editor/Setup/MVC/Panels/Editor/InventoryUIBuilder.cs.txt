#if HAS_INVENTORY_SYSTEM && HAS_UNIMVC && HAS_EVENT_SYSTEM && HAS_INPUT_SYSTEM
using InventorySystem.UI;
using TMPro;
using UniMVC;
using UnityEditor;
using UnityEngine;
using UnityEngine.InputSystem.UI;
using UnityEngine.UI;

namespace InventorySystem.UIEditor
{
    /// <summary>
    /// <c>GameObject &gt; UI &gt; Inventory System</c>: builds a ready-to-use, fully wired inventory UI under
    /// the selected (or first, or a new) canvas, with a plain dark look to restyle freely. It adds a
    /// <see cref="UIManager"/> (and an <see cref="InventoryController"/>) to that canvas if it has none, and
    /// lists the new panels in it (and the window's own views in the window), so <c>UIManager.Initialize</c>
    /// initializes all of them:
    /// <list type="bullet">
    /// <item><b>Inventory Window</b>: an <see cref="InventoryPanel"/> with its header, weight bar,
    /// <see cref="InventoryListPanel"/> and <see cref="InventoryGridPanel"/>, plus an <see cref="InventoryTooltipPanel"/>.</item>
    /// <item><b>Inventory Notifications</b>: an <see cref="InventoryNotificationPanel"/>.</item>
    /// </list>
    /// </summary>
    internal static class InventoryUIBuilder
    {
        private const string MenuRoot = "GameObject/UI/Inventory System/";
        private const int UILayer = 5;

        private static readonly Color PanelColor = new(0.09f, 0.1f, 0.12f, 0.94f);
        private static readonly Color SlotColor = new(1f, 1f, 1f, 0.12f);
        private static readonly Color CellColor = new(1f, 1f, 1f, 0.06f);
        private static readonly Color GridItemColor = new(0.35f, 0.55f, 0.9f, 0.35f);
        private static readonly Color BarBackColor = new(1f, 1f, 1f, 0.1f);
        private static readonly Color BarFillColor = new(0.95f, 0.75f, 0.3f, 1f);
        private static readonly Color MutedTextColor = new(0.75f, 0.77f, 0.8f, 1f);
        private static readonly Color Transparent = new(0f, 0f, 0f, 0f);

        private static Sprite RoundedSprite => AssetDatabase.GetBuiltinExtraResource<Sprite>("UI/Skin/Background.psd");

        [MenuItem(MenuRoot + "Inventory Window", false, 2100)]
        private static void CreateWindow(MenuCommand command)
        {
            var canvas = PrepareCanvas(command.context as GameObject);
            var window = BuildWindow(canvas.transform);
            var tooltip = BuildTooltip(canvas.transform);
            List(canvas, window.GetComponent<InventoryPanel>(), tooltip.GetComponent<InventoryTooltipPanel>());

            Undo.RegisterCreatedObjectUndo(window, "Create Inventory Window");
            Undo.RegisterCreatedObjectUndo(tooltip, "Create Inventory Window");
            Selection.activeGameObject = window;
        }

        [MenuItem(MenuRoot + "Inventory Notifications", false, 2101)]
        private static void CreateNotifications(MenuCommand command)
        {
            var canvas = PrepareCanvas(command.context as GameObject);
            var notifications = BuildNotifications(canvas.transform);
            List(canvas, notifications.GetComponent<InventoryNotificationPanel>());

            Undo.RegisterCreatedObjectUndo(notifications, "Create Inventory Notifications");
            Selection.activeGameObject = notifications;
        }

        #region Window

        private static GameObject BuildWindow(Transform parent)
        {
            var panel = NewRect("Inventory Panel", parent);
            panel.anchorMin = panel.anchorMax = panel.pivot = new Vector2(0.5f, 0.5f);
            AddImage(panel, RoundedSprite, PanelColor, true);
            var layout = panel.gameObject.AddComponent<VerticalLayoutGroup>();
            layout.padding = new RectOffset(18, 18, 14, 18);
            layout.spacing = 10f;
            layout.childControlWidth = layout.childControlHeight = true;
            layout.childForceExpandWidth = layout.childForceExpandHeight = false;
            Fit(panel, ContentSizeFitter.FitMode.PreferredSize, ContentSizeFitter.FitMode.PreferredSize);
            var window = panel.gameObject.AddComponent<InventoryPanel>();

            // Header: title, capacity, close button.
            var header = NewRect("Header", panel);
            var headerLayout = header.gameObject.AddComponent<HorizontalLayoutGroup>();
            headerLayout.spacing = 12f;
            headerLayout.childAlignment = TextAnchor.MiddleLeft;
            headerLayout.childControlWidth = headerLayout.childControlHeight = true;
            headerLayout.childForceExpandWidth = headerLayout.childForceExpandHeight = false;
            Layout(header, minHeight: 34f);

            var title = AddText(NewRect("Title", header), "Inventory", 26f, TextAlignmentOptions.Left, FontStyles.Bold, Color.white);
            Layout((RectTransform)title.transform, flexibleWidth: 1f);
            var capacity = AddText(NewRect("Capacity", header), "0 / 0", 20f, TextAlignmentOptions.Right, FontStyles.Normal, MutedTextColor);

            var close = NewRect("Close Button", header);
            var closeImage = AddImage(close, RoundedSprite, new Color(1f, 1f, 1f, 0.12f), true);
            close.gameObject.AddComponent<Button>().targetGraphic = closeImage;
            var closeButton = close.gameObject.AddComponent<InventoryCloseButton>();
            Layout(close, preferredWidth: 30f, preferredHeight: 30f);
            Stretch((RectTransform)AddText(NewRect("Label", close), "X", 18f, TextAlignmentOptions.Center, FontStyles.Bold, Color.white).transform);

            // Weight bar (Weight mode only; the panel hides it otherwise).
            var bar = NewRect("Weight Bar", panel);
            AddImage(bar, RoundedSprite, BarBackColor, false);
            Layout(bar, preferredHeight: 8f, flexibleWidth: 1f);
            var fill = NewRect("Fill", bar);
            Stretch(fill);
            fill.anchorMax = new Vector2(0.4f, 1f);
            AddImage(fill, RoundedSprite, BarFillColor, false);

            var list = BuildListContent(panel);
            var grid = BuildGridContent(panel);

            var so = new SerializedObject(window);
            var contents = so.FindProperty("contents");
            contents.arraySize = 2;
            contents.GetArrayElementAtIndex(0).objectReferenceValue = list;
            contents.GetArrayElementAtIndex(1).objectReferenceValue = grid;
            so.FindProperty("capacityText").objectReferenceValue = capacity;
            so.FindProperty("weightBar").objectReferenceValue = bar.gameObject;
            so.FindProperty("weightFill").objectReferenceValue = fill;
            so.ApplyModifiedPropertiesWithoutUndo();

            // The window initializes its own views: both content panels and the close button.
            window.ChildViews.Add(list);
            window.ChildViews.Add(grid);
            window.ChildViews.Add(closeButton);

            return panel.gameObject;
        }

        private static InventoryListPanel BuildListContent(RectTransform parent)
        {
            var root = NewRect("List Content", parent);
            AddImage(root, null, Transparent, true); // drop target between slots
            var layout = Layout(root, preferredWidth: 400f, preferredHeight: 300f);
            var scroll = root.gameObject.AddComponent<ScrollRect>();
            scroll.horizontal = false;
            scroll.movementType = ScrollRect.MovementType.Clamped;
            scroll.scrollSensitivity = 30f;

            var viewport = NewRect("Viewport", root);
            Stretch(viewport);
            viewport.gameObject.AddComponent<RectMask2D>();

            var content = NewRect("Content", viewport);
            content.anchorMin = new Vector2(0f, 1f);
            content.anchorMax = new Vector2(1f, 1f);
            content.pivot = new Vector2(0.5f, 1f);
            content.sizeDelta = Vector2.zero;
            var gridLayout = content.gameObject.AddComponent<GridLayoutGroup>();
            gridLayout.cellSize = new Vector2(72f, 72f);
            gridLayout.spacing = new Vector2(6f, 6f);
            gridLayout.constraint = GridLayoutGroup.Constraint.FixedColumnCount;
            gridLayout.constraintCount = 6;
            Fit(content, ContentSizeFitter.FitMode.Unconstrained, ContentSizeFitter.FitMode.PreferredSize);

            scroll.viewport = viewport;
            scroll.content = content;

            var template = BuildSlot("Slot Template", root, false, SlotColor);
            template.gameObject.SetActive(false);

            var panel = root.gameObject.AddComponent<InventoryListPanel>();
            var so = new SerializedObject(panel);
            so.FindProperty("slotTemplate").objectReferenceValue = template;
            so.FindProperty("content").objectReferenceValue = gridLayout;
            so.FindProperty("layoutElement").objectReferenceValue = layout;
            so.ApplyModifiedPropertiesWithoutUndo();
            return panel;
        }

        private static InventoryGridPanel BuildGridContent(RectTransform parent)
        {
            var root = NewRect("Grid Content", parent);
            AddImage(root, null, Transparent, true); // drop target for empty cells
            var layout = Layout(root, preferredWidth: 400f, preferredHeight: 300f);

            var cells = NewRect("Cells", root);
            var items = NewRect("Items", root);
            var preview = NewRect("Preview", root);

            var cellTemplate = NewRect("Cell Template", root);
            var cellImage = AddImage(cellTemplate, RoundedSprite, CellColor, false);
            cellTemplate.gameObject.SetActive(false);

            var itemTemplate = BuildSlot("Item Template", root, true, GridItemColor);
            itemTemplate.gameObject.SetActive(false);

            var panel = root.gameObject.AddComponent<InventoryGridPanel>();
            var so = new SerializedObject(panel);
            so.FindProperty("cellLayer").objectReferenceValue = cells;
            so.FindProperty("itemLayer").objectReferenceValue = items;
            so.FindProperty("previewLayer").objectReferenceValue = preview;
            so.FindProperty("cellTemplate").objectReferenceValue = cellImage;
            so.FindProperty("itemTemplate").objectReferenceValue = itemTemplate;
            so.FindProperty("layoutElement").objectReferenceValue = layout;
            so.ApplyModifiedPropertiesWithoutUndo();
            return panel;
        }

        private static InventorySlotButton BuildSlot(string name, Transform parent, bool gridItem, Color color)
        {
            var root = NewRect(name, parent);
            root.sizeDelta = new Vector2(72f, 72f);
            var background = AddImage(root, RoundedSprite, color, true);

            var button = root.gameObject.AddComponent<Button>();
            button.targetGraphic = background;
            var colors = button.colors;
            colors.normalColor = new Color(1f, 1f, 1f, 0.7f);
            colors.highlightedColor = Color.white;
            colors.pressedColor = new Color(1f, 1f, 1f, 0.85f);
            colors.selectedColor = new Color(1f, 1f, 1f, 0.7f);
            colors.fadeDuration = 0.06f;
            button.colors = colors;
            button.navigation = new Navigation { mode = Navigation.Mode.None };
            root.gameObject.AddComponent<CanvasGroup>();

            var iconRect = NewRect("Icon", root);
            if (gridItem)
            {
                iconRect.anchorMin = iconRect.anchorMax = new Vector2(0.5f, 0.5f);
                iconRect.sizeDelta = new Vector2(52f, 52f);
            }
            else
            {
                Stretch(iconRect, 9f);
            }

            var icon = AddImage(iconRect, null, Color.white, false);
            icon.preserveAspect = true;

            var amountRect = NewRect("Amount", root);
            Stretch(amountRect, 5f);
            var amount = AddText(amountRect, string.Empty, 18f, TextAlignmentOptions.BottomRight, FontStyles.Bold, Color.white);

            var slot = root.gameObject.AddComponent<InventorySlotButton>();
            var so = new SerializedObject(slot);
            so.FindProperty("icon").objectReferenceValue = icon;
            so.FindProperty("amountText").objectReferenceValue = amount;
            so.ApplyModifiedPropertiesWithoutUndo();
            return slot;
        }

        private static GameObject BuildTooltip(Transform parent)
        {
            var panel = NewRect("Inventory Tooltip", parent);
            panel.anchorMin = panel.anchorMax = new Vector2(0.5f, 0.5f);
            panel.pivot = new Vector2(0f, 1f);
            AddImage(panel, RoundedSprite, new Color(0.05f, 0.05f, 0.07f, 0.96f), false);
            var group = panel.gameObject.AddComponent<CanvasGroup>();
            group.blocksRaycasts = false;
            group.interactable = false;
            var layout = panel.gameObject.AddComponent<VerticalLayoutGroup>();
            layout.padding = new RectOffset(12, 12, 8, 10);
            layout.spacing = 2f;
            layout.childControlWidth = layout.childControlHeight = true;
            layout.childForceExpandWidth = layout.childForceExpandHeight = false;
            Fit(panel, ContentSizeFitter.FitMode.PreferredSize, ContentSizeFitter.FitMode.PreferredSize);

            var nameText = AddText(NewRect("Name", panel), "Item", 20f, TextAlignmentOptions.Left, FontStyles.Bold, Color.white);
            var detailText = AddText(NewRect("Details", panel), string.Empty, 16f, TextAlignmentOptions.Left, FontStyles.Normal, MutedTextColor);

            var tooltip = panel.gameObject.AddComponent<InventoryTooltipPanel>();
            var so = new SerializedObject(tooltip);
            so.FindProperty("nameText").objectReferenceValue = nameText;
            so.FindProperty("detailText").objectReferenceValue = detailText;
            so.ApplyModifiedPropertiesWithoutUndo();

            panel.gameObject.SetActive(false);
            return panel.gameObject;
        }

        #endregion

        #region Notifications

        private static GameObject BuildNotifications(Transform parent)
        {
            var root = NewRect("Inventory Notifications", parent);
            root.anchorMin = root.anchorMax = root.pivot = Vector2.zero;
            root.anchoredPosition = new Vector2(24f, 140f);
            root.sizeDelta = new Vector2(380f, 0f);
            var layout = root.gameObject.AddComponent<VerticalLayoutGroup>();
            layout.spacing = 6f;
            layout.childAlignment = TextAnchor.LowerLeft;
            layout.childControlWidth = layout.childControlHeight = true;
            layout.childForceExpandWidth = layout.childForceExpandHeight = false;
            Fit(root, ContentSizeFitter.FitMode.Unconstrained, ContentSizeFitter.FitMode.PreferredSize);
            var notifications = root.gameObject.AddComponent<InventoryNotificationPanel>();

            var toastRect = NewRect("Toast Template", root);
            AddImage(toastRect, RoundedSprite, new Color(0.05f, 0.05f, 0.07f, 0.85f), false);
            toastRect.gameObject.AddComponent<CanvasGroup>().blocksRaycasts = false;
            var toastLayout = toastRect.gameObject.AddComponent<HorizontalLayoutGroup>();
            toastLayout.padding = new RectOffset(8, 12, 6, 6);
            toastLayout.spacing = 10f;
            toastLayout.childAlignment = TextAnchor.MiddleLeft;
            toastLayout.childControlWidth = toastLayout.childControlHeight = true;
            toastLayout.childForceExpandWidth = toastLayout.childForceExpandHeight = false;
            Layout(toastRect, minHeight: 44f);

            var iconRect = NewRect("Icon", toastRect);
            var icon = AddImage(iconRect, null, Color.white, false);
            icon.preserveAspect = true;
            Layout(iconRect, preferredWidth: 32f, preferredHeight: 32f);

            var message = AddText(NewRect("Message", toastRect), string.Empty, 20f, TextAlignmentOptions.Left, FontStyles.Normal, Color.white);
            Layout((RectTransform)message.transform, flexibleWidth: 1f);

            var toast = toastRect.gameObject.AddComponent<InventoryToastPanel>();
            var toastSo = new SerializedObject(toast);
            toastSo.FindProperty("icon").objectReferenceValue = icon;
            toastSo.FindProperty("message").objectReferenceValue = message;
            toastSo.ApplyModifiedPropertiesWithoutUndo();
            toastRect.gameObject.SetActive(false);

            var so = new SerializedObject(notifications);
            so.FindProperty("toastTemplate").objectReferenceValue = toast;
            so.FindProperty("container").objectReferenceValue = root;
            so.ApplyModifiedPropertiesWithoutUndo();

            return root.gameObject;
        }

        #endregion

        #region Helpers

        // Adds the new top-level views to the canvas's UIManager lists.
        private static void List(Canvas canvas, params ViewBase[] views)
        {
            var ui = canvas.GetComponent<UIManager>();
            Undo.RecordObject(ui, "List Inventory Views");
            foreach (var view in views)
            {
                ui.Views.Add(view);
            }

            EditorUtility.SetDirty(ui);
        }

        // The selected (or first, or a new) canvas, with a UIManager and an EventSystem in the scene.
        private static Canvas PrepareCanvas(GameObject context)
        {
            var canvas = context != null ? context.GetComponentInParent<Canvas>() : null;
            if (canvas == null)
            {
                canvas = Object.FindFirstObjectByType<Canvas>();
            }

            if (canvas != null)
            {
                canvas = canvas.rootCanvas;
            }
            else
            {
                var go = new GameObject("Canvas", typeof(RectTransform), typeof(Canvas), typeof(CanvasScaler), typeof(GraphicRaycaster))
                {
                    layer = UILayer
                };
                canvas = go.GetComponent<Canvas>();
                canvas.renderMode = RenderMode.ScreenSpaceOverlay;
                var scaler = go.GetComponent<CanvasScaler>();
                scaler.uiScaleMode = CanvasScaler.ScaleMode.ScaleWithScreenSize;
                scaler.referenceResolution = new Vector2(1920f, 1080f);
                scaler.matchWidthOrHeight = 0.5f;
                Undo.RegisterCreatedObjectUndo(go, "Create Canvas");
            }

            if (!canvas.TryGetComponent<UIManager>(out var ui))
            {
                ui = Undo.AddComponent<UIManager>(canvas.gameObject);
                Debug.Log($"[InventoryUIBuilder] Added a UIManager to '{canvas.name}'. Call its Initialize() from your bootstrap code.", canvas);
            }

            // The inventory's events reach the UI through this controller, on the always-active UIManager object.
            if (!canvas.TryGetComponent<InventoryController>(out var controller))
            {
                controller = Undo.AddComponent<InventoryController>(canvas.gameObject);
            }

            Undo.RecordObject(ui, "Add Inventory Controller");
            ui.AddController(controller);
            EditorUtility.SetDirty(ui);

            if (Object.FindFirstObjectByType<UnityEngine.EventSystems.EventSystem>() == null)
            {
                var eventSystem = new GameObject("EventSystem", typeof(UnityEngine.EventSystems.EventSystem), typeof(InputSystemUIInputModule));
                Undo.RegisterCreatedObjectUndo(eventSystem, "Create EventSystem");
            }

            return canvas;
        }

        private static RectTransform NewRect(string name, Transform parent)
        {
            var go = new GameObject(name, typeof(RectTransform)) { layer = UILayer };
            var rect = (RectTransform)go.transform;
            rect.SetParent(parent, false);
            return rect;
        }

        private static void Stretch(RectTransform rect, float inset = 0f)
        {
            rect.anchorMin = Vector2.zero;
            rect.anchorMax = Vector2.one;
            rect.offsetMin = new Vector2(inset, inset);
            rect.offsetMax = new Vector2(-inset, -inset);
        }

        private static Image AddImage(RectTransform rect, Sprite sprite, Color color, bool raycastTarget)
        {
            var image = rect.gameObject.AddComponent<Image>();
            image.sprite = sprite;
            image.type = sprite != null ? Image.Type.Sliced : Image.Type.Simple;
            image.color = color;
            image.raycastTarget = raycastTarget;
            return image;
        }

        private static TextMeshProUGUI AddText(RectTransform rect, string text, float size, TextAlignmentOptions alignment,
            FontStyles style, Color color)
        {
            var label = rect.gameObject.AddComponent<TextMeshProUGUI>();
            label.text = text;
            label.fontSize = size;
            label.alignment = alignment;
            label.fontStyle = style;
            label.color = color;
            label.raycastTarget = false;
            label.textWrappingMode = TextWrappingModes.NoWrap;
            return label;
        }

        private static LayoutElement Layout(RectTransform rect, float minHeight = -1f, float preferredWidth = -1f,
            float preferredHeight = -1f, float flexibleWidth = -1f)
        {
            var element = rect.gameObject.AddComponent<LayoutElement>();
            element.minHeight = minHeight;
            element.preferredWidth = preferredWidth;
            element.preferredHeight = preferredHeight;
            element.flexibleWidth = flexibleWidth;
            return element;
        }

        private static void Fit(RectTransform rect, ContentSizeFitter.FitMode horizontal, ContentSizeFitter.FitMode vertical)
        {
            var fitter = rect.gameObject.AddComponent<ContentSizeFitter>();
            fitter.horizontalFit = horizontal;
            fitter.verticalFit = vertical;
        }

        #endregion
    }
