#if HAS_INVENTORY_SYSTEM && HAS_UNIMVC && HAS_EVENT_SYSTEM && HAS_INPUT_SYSTEM
using UniMVC;

namespace InventorySystem.UI
{
    /// <summary>
    /// Uses the item selected in the <see cref="InventoryPanel"/> it sits in - for now only by raising the panel's
    /// <see cref="InventoryPanel.UseRequested"/>, for your code to do what "use" means (eat, equip, read...).
    /// Nothing happens while nothing listens. Can't be pressed while nothing is selected.
    /// </summary>
    public sealed class InventoryUseButton : ButtonViewBase
    {
        private InventoryPanel _panel;

        protected override void OnInitialize()
        {
            base.OnInitialize();
            _panel = GetComponentInParent<InventoryPanel>(true);
            if (_panel != null)
            {
                _panel.SelectionChanged += Refresh;
            }

            Refresh();
        }

        private void OnDestroy()
        {
            if (_panel != null)
            {
                _panel.SelectionChanged -= Refresh;
            }
        }

        protected override void OnClick()
        {
            if (_panel != null)
            {
                _panel.UseSelected();
            }
        }

        private void Refresh() => Interactable = _panel != null && _panel.HasSelection;
    }
}
#endif
