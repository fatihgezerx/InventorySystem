#if HAS_INVENTORY_SYSTEM && HAS_UNIMVC && HAS_EVENT_SYSTEM && HAS_INPUT_SYSTEM
#if HAS_BASIC_TOOLTIP
using BasicTooltip;
#endif
using TMPro;
using UniMVC;
using UnityEngine;
using UnityEngine.EventSystems;
using UnityEngine.UI;

namespace InventorySystem.UI
{
    /// <summary>
    /// One inventory slot: a button with an icon and an amount label. Holds no inventory logic - every
    /// pointer event (click, hover, drag, drop) goes to the <see cref="InventoryContentPanel"/> that
    /// created it, which knows which slot it shows. Left click comes through the Button (so it gets the
    /// Button's hover / pressed look); other buttons through <see cref="OnPointerClick"/>. With Basic Tooltip, it
    /// gives its item's tooltip to a Tooltip Trigger on it (Easy UI's Shows Tooltip role).
    /// </summary>
    [RequireComponent(typeof(CanvasGroup))]
    public sealed class InventorySlotButton : ButtonViewBase,
        IPointerClickHandler, IBeginDragHandler, IDragHandler, IEndDragHandler, IDropHandler, ISelectHandler
#if HAS_BASIC_TOOLTIP
        , ITooltipSource
#endif
    {
        [HideInInspector] [SerializeField] private Image icon;

        [Tooltip("Shows the amount when it is more than 1.")]
        [HideInInspector] [SerializeField] private TMP_Text amountText;

        [Tooltip("Opacity of the slot while its item is being dragged.")]
        [Range(0f, 1f)] [SerializeField] private float draggedAlpha = 0.35f;

        // Opacity of a grid item held by the pointer: the tinted cells it hovers over show through.
        private const float LiftedAlpha = 0.7f;

        // Sorts the slot above everything else while it is held or settling - added on the first time.
        private const int LiftedSortingOrder = 30000;

        private CanvasGroup _group;
        private RectTransform _rect;
        private Canvas _liftCanvas;
        private bool _hasItem;
        private bool _receiving;

        /// <summary>The panel this slot belongs to. Set by the panel.</summary>
        internal InventoryContentPanel Owner { get; set; }

        /// <summary>The slot index this button shows. Set by the panel.</summary>
        public int SlotIndex { get; internal set; }

        public RectTransform RectTransform => _rect != null ? _rect : _rect = (RectTransform)transform;

        private CanvasGroup Group => _group != null ? _group : _group = GetComponent<CanvasGroup>();

        /// <summary>
        /// Draws <paramref name="slot"/>; an empty slot (or null) shows only the background, and can't be pressed
        /// (nor dragged: the panel ignores a drag from an empty slot).
        /// </summary>
        public void Show(InventorySlot slot)
        {
            if (slot == null || slot.IsEmpty)
            {
                ShowEmpty();
                return;
            }

            SetPressable(true);
            icon.sprite = slot.Item.Icon;
            icon.enabled = icon.sprite != null;

            if (slot.Amount > 1)
            {
                amountText.SetText("{0}", slot.Amount);
            }
            else
            {
                amountText.SetText(string.Empty);
            }
        }

        public void ShowEmpty()
        {
            SetPressable(false);
            icon.sprite = null;
            icon.enabled = false;
            amountText.SetText(string.Empty);
        }

        // Pressable only while it holds an item. Things still drop onto an empty slot.
        private void SetPressable(bool pressable)
        {
            _hasItem = pressable;
            ApplyInteractable();
        }

        /// <summary>
        /// A gamepad player carries an item: even an empty slot can be selected then, to put it down there. Off again
        /// when it is put down.
        /// </summary>
        internal void SetReceiving(bool receiving)
        {
            if (_receiving == receiving)
            {
                return;
            }

            _receiving = receiving;
            ApplyInteractable();
        }

        private void ApplyInteractable()
        {
            var interactable = _hasItem || _receiving;
            if (IsInitialized && Interactable != interactable)
            {
                Interactable = interactable;
            }
        }

        /// <summary>Whether the d-pad / stick can move the selection from or to this slot (automatic navigation).</summary>
        internal void SetNavigable(bool navigable)
        {
            if (!IsInitialized)
            {
                return;
            }

            var navigation = Button.navigation;
            navigation.mode = navigable ? Navigation.Mode.Automatic : Navigation.Mode.None;
            Button.navigation = navigation;
        }

        /// <summary>
        /// Grid items: sizes the icon to the item's unrotated footprint and turns it
        /// <paramref name="rotation"/> quarter turns clockwise, so the sprite keeps its proportions.
        /// </summary>
        public void SetIconRotation(int rotation, Vector2 unrotatedSize)
        {
            var iconRect = icon.rectTransform;
            iconRect.anchorMin = iconRect.anchorMax = new Vector2(0.5f, 0.5f);
            iconRect.anchoredPosition = Vector2.zero;
            iconRect.sizeDelta = unrotatedSize;
            iconRect.localEulerAngles = new Vector3(0f, 0f, -90f * rotation);
        }

        /// <summary>Fades the slot and lets raycasts pass through it while its item is dragged.</summary>
        internal void SetDragged(bool dragged)
        {
            Group.alpha = dragged ? draggedAlpha : 1f;
            Group.blocksRaycasts = !dragged;
        }

        /// <summary>
        /// Grid items: draws the slot above every other element while the pointer holds it and while it settles,
        /// nearly opaque and never in the way of the pointer. Off again once it is placed (its opacity is restored
        /// when its drag ends).
        /// </summary>
        /// <remarks>
        /// It sorts through a Canvas of its own (with a GraphicRaycaster, so it can still be clicked and dragged),
        /// added the first time; that Canvas stays, and only its Override Sorting is switched - disabling a Canvas
        /// would hide everything in it.
        /// </remarks>
        internal void SetLifted(bool lifted)
        {
            if (lifted && _liftCanvas == null)
            {
                _liftCanvas = gameObject.AddComponent<Canvas>();
                gameObject.AddComponent<GraphicRaycaster>();
            }

            if (_liftCanvas == null)
            {
                return;
            }

            _liftCanvas.overrideSorting = lifted;
            if (lifted)
            {
                _liftCanvas.sortingOrder = LiftedSortingOrder;
                Group.alpha = LiftedAlpha;
                Group.blocksRaycasts = false;
            }
        }

        protected override void OnClick()
        {
            if (Owner != null)
            {
                Owner.OnSlotClicked(this, PointerEventData.InputButton.Left);
            }
        }

        public void OnSelect(BaseEventData eventData)
        {
            if (Owner != null)
            {
                Owner.OnSlotFocused(this);
            }
        }

        public void OnPointerClick(PointerEventData eventData)
        {
            if (Owner != null && eventData.button != PointerEventData.InputButton.Left)
            {
                Owner.OnSlotClicked(this, eventData.button);
            }
        }

#if HAS_BASIC_TOOLTIP
        bool ITooltipSource.TryGetTooltip(out TooltipContent content)
        {
            content = default;
            return Owner != null && Owner.TryGetTooltip(this, out content);
        }
#endif

        public void OnBeginDrag(PointerEventData eventData)
        {
            if (Owner != null)
            {
                Owner.OnSlotBeginDrag(this, eventData);
            }
            else
            {
                eventData.pointerDrag = null;
            }
        }

        public void OnDrag(PointerEventData eventData)
        {
            if (Owner != null)
            {
                Owner.OnSlotDrag(eventData);
            }
        }

        public void OnEndDrag(PointerEventData eventData)
        {
            if (Owner != null)
            {
                Owner.OnSlotEndDrag(eventData);
            }
        }

        public void OnDrop(PointerEventData eventData)
        {
            if (Owner != null)
            {
                Owner.OnSlotDrop(this, eventData);
            }
        }
    }
}
#endif
